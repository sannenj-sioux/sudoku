bool CCommand::execute(int nInputValue) {
  if (_pOwner == nullptr) {
    return false;
  }

  _stPoint = _pOwner->getCurPoint();
  return _pOwner->setCurValue(nInputValue, _nPreValue);
}

void CCommand::undo() {
  if (_pOwner != nullptr) {
    _pOwner->setPointValue(_stPoint, _nPreValue);
  }
