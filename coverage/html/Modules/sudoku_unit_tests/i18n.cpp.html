// Default English
I18n::I18n() : dict_(&ENGLISH_DICT) {}

I18n& I18n::Instance() {
  static I18n i18n;
  return i18n;
}

void I18n::SetLanguage(Language language) {
  switch (language) {
    case Language::ENGLISH:
      dict_ = &ENGLISH_DICT;
      break;
    case Language::CHINESE:
      dict_ = &CHINESE_DICT;
      break;
    case Language::MAX:
    default:
      assert(false);
  }
}

std::string I18n::Get(Key key) const {
  assert(dict_->count(key));
  return dict_->at(key);
}
