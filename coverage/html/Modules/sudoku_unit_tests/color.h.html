class Modifier {
  Code md, bg, fg;

 public:
  Modifier() : md(Color::RESET), bg(Color::BG_DEFAULT), fg(Color::FG_DEFAULT) {}
  Modifier(Code Mode, Code BackGround, Code FrontGround)
      : md(Mode), bg(BackGround), fg(FrontGround) {}
  friend std::ostream& operator<<(std::ostream& output_stream, const Modifier& mod) {
    return output_stream << "\033[" << mod.md << ";" << mod.bg << ";" << mod.fg << "m";
  }
};
